4. Exemplo Completo de Implementação 
import requests
import time

class CrossXMLAPI:
    def __init__(self, api_key, api_secret, base_url="https://crossxml.com.br/api/v1"):
        self.base_url = base_url
        self.headers = {
            'Content-Type': 'application/json',
            'X-API-Key': api_key,
            'X-API-Secret': api_secret
        }
    
    def create_job(self, invoice_keys):
        """Criar novo job de download"""
        response = requests.post(f"{self.base_url}/jobs", 
                               json={"invoice_keys": invoice_keys}, 
                               headers=self.headers)
        return response.json()
    
    def get_job_status(self, job_id):
        """Obter status do job"""
        response = requests.get(f"{self.base_url}/jobs/{job_id}", 
                              headers=self.headers)
        return response.json()
    
    def download_xml(self, job_id, filename, save_path="."):
        """Baixar arquivo XML específico"""
        response = requests.get(f"{self.base_url}/jobs/{job_id}/files/{filename}", 
                              headers=self.headers)
        
        file_path = f"{save_path}/{filename}"
        with open(file_path, 'wb') as f:
            f.write(response.content)
        return file_path

class CrossXMLAPI {
    constructor(apiKey, apiSecret, baseUrl = "https://crossxml.com.br/api/v1") {
        this.baseUrl = baseUrl;
        this.headers = {
            'Content-Type': 'application/json',
            'X-API-Key': apiKey,
            'X-API-Secret': apiSecret
        };
    }

    async criarJob(chavesXml) {
        const response = await fetch(`${this.baseUrl}/jobs`, {
            method: 'POST',
            headers: this.headers,
            body: JSON.stringify({ invoice_keys: chavesXml })
        });
        return await response.json();
    }

    async statusJob(jobId) {
        const response = await fetch(`${this.baseUrl}/jobs/${jobId}`, {
            headers: this.headers
        });
        return await response.json();
    }

    async baixarXml(jobId, nomeArquivo) {
        const response = await fetch(`${this.baseUrl}/jobs/${jobId}/files/${nomeArquivo}`, {
            headers: this.headers
        });
        return await response.blob();
    }
}

// Exemplo de uso
const api = new CrossXMLAPI(
    "api_xxxxxxxxxxxxxxxx",
    "secret_xxxxxxxxxxxxxxxx"
);

async function exemploDownload() {
    // 1. Criar job
    const job = await api.criarJob(["35250513516247000107550010000113401146202508"]);
    console.log("Job criado:", job.job_id);

    // 2. Aguardar conclusão
    let status;
    do {
        status = await api.statusJob(job.job_id);
        console.log(`Status: ${status.status} - ${status.progress.percentage}%`);
        await new Promise(resolve => setTimeout(resolve, 2000));
    } while (status.status === "running");

    // 3. Baixar arquivos
    if (status.status === "completed") {
        for (const arquivo of status.files) {
            const blob = await api.baixarXml(job.job_id, arquivo.filename);
            console.log(`XML baixado: ${arquivo.filename}`);
        }
    }
}
